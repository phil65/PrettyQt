class ElapsedTimer(QtCore.QElapsedTimer):
    """Fast way to calculate elapsed times."""

    def __bool__(self):
        return self.isValid()

    def get_clock_type(self) -> ClockTypeStr:
        """Return current clock type.

        Returns:
            clock type
        """
        return CLOCK_TYPE.inverse[self.clockType()]
